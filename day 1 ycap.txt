#include<stdio.h>
int main(){
   char c1,c2;
   int i1,i2;
   float f1,f2;
   c1='h';
   i1=80.56;
   f1=12.6;
   c2=i1;
   i2=f1;
   printf("\nc2=%c,i2=%d",c2,i2);
